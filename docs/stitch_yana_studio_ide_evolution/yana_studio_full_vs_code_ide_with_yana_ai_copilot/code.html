import { GovernanceEngine, PermissionBoundary } from "../permissions";
import { AgentTask, ExecutionContext, TaskResult } from "../agents/types";
import { AuditLogger } from "../permissions/audit_logger";
/**
* Yana Runtime Orchestrator: Deterministic execution boundary with human-in-the-loop governance.
*/
export class RuntimeOrchestrator {
private readonly policyEnforcer: GovernanceEngine;
private readonly logger: AuditLogger;
constructor(boundary: PermissionBoundary) {
this.policyEnforcer = new GovernanceEngine(boundary, { strictMode: true });
this.logger = new AuditLogger("runtime.core.orchestrator");
}
public async executeWithGovernance(task: AgentTask, context: ExecutionContext): Promise<TaskResult> {
const evaluation = await this.policyEnforcer.evaluate(task.action);
if (evaluation.requiresHumanApproval) {
await this.logger.warn(`Interception: Task ${task.id} triggered gate.`);
return this.suspendForApproval(task, evaluation.requestedCapability);
}
if (!evaluation.allowed) {
throw new SecurityException(`Operation denied by policy: ${evaluation.reason}`);
}
const worker = await context.acquireWorker();
return await worker.dispatch(task);
}
}
